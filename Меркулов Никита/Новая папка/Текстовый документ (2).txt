const int LIGHT_PIN = A0; 
const int LED_PIN1 = 8; 
const int LED_PIN2 = 9; 
const int LED_PIN3 = 10; 

const unsigned long READ_MS = 200; 

unsigned long blink1_ms = 500;
unsigned long blink2_ms = 200;
unsigned long blink3_ms = 800;

unsigned long lastRead = 0; 
unsigned long lastBlink1 = 0; 
unsigned long lastBlink2 = 0; 
unsigned long lastBlink3 = 0; 

bool led1On = false; 
bool led2On = false; 
bool led3On = false; 

void setup() { 
  pinMode(LED_PIN1, OUTPUT); 
  pinMode(LED_PIN2, OUTPUT); 
  pinMode(LED_PIN3, OUTPUT); 
  Serial.begin(9600); 
} 

void loop() { 
  unsigned long now = millis(); 

  if (now - lastRead >= READ_MS) { 
    lastRead = now; 
    int light = analogRead(LIGHT_PIN); 
    Serial.println(light); 

    blink1_ms = map(light, 0, 1023, 1000, 200); 
    blink2_ms = map(light, 0, 1023, 600, 100); 
    blink3_ms = map(light, 0, 1023, 1500, 400); 
  } 

  if (now - lastBlink1 >= blink1_ms) { 
    lastBlink1 = now; 
    led1On = !led1On; 
    digitalWrite(LED_PIN1, led1On); 
  } 

  if (now - lastBlink2 >= blink2_ms) { 
    lastBlink2 = now; 
    led2On = !led2On; 
    digitalWrite(LED_PIN2, led2On); 
  } 

  if (now - lastBlink3 >= blink3_ms) { 
    lastBlink3 = now; 
    led3On = !led3On; 
    digitalWrite(LED_PIN3, led3On); 
  } 
}
